  become: yes
  tasks:
    - name: create user

        name: vaidikmaheshwari
        state: present
        shell: /bin/bash
    - name: set permission for user
      file:
        path: /home/vaidikmaheshwari
        owner: vaidikmaheshwari
        group: vaidikmaheshwari
