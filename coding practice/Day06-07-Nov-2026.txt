## ansible install docker
```yaml
- name: install docker
  hosts: all
  become: yes
  tasks:
    - name: install docker
      yum:
        name: docker
        state: present
    - name: start docker
      service:
        name: docker
        state: started
        enabled: yes

kubernetes - pod with volume mountPath
apiVersion: v1
kind: Pod
metadata:
  name: vaidik-pod
spec: 
  containers:
    - name: web
      image: nginx
      volumeMounts:
        - mountPath: /usr/share/nginx/html
          name: html-Volume
  volumes:
    - name: html-Volume
      hostPath:
        path: /data
        type: Directory
  


shell script - check for running device
#!/bin/bash
services=("nginx" "docker" "httpd")
for service in "${services[@]}"
do 
  if systemctl is-active --quiet $service; then
     echo "$service is running"
  else 
     echo "$service is not running"
  fi
done
